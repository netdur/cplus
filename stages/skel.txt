// A str field is only sound while its bytes outlive the struct. Literals do;
// a computed key ("row_${i}") does not, and the borrow check does NOT catch it
// when the view arrives as a parameter — verified: it compiles clean and ASan
// reports heap-use-after-free on the first read. Interning fixes it without
// giving up str: the table owns the bytes for the process, so the stored view
// is permanently valid, and key comparison in slot() becomes a pointer compare
// instead of a memcmp.

// str when the value never changes, Text when it does. Every prop here has a
// setter, so every string is owned; `key` above is the build-time-only case.
struct ButtonProps {
    label: text::Text,                  // Button.Text
    text_color: vocab::Color,           // Button.TextColor
    corner_radius: vocab::Corners,      // Button.CornerRadius
    font_style: vocab::FontStyle,       // Button.FontAttributes
    font_size: f64,                     // Button.FontSize
    line_break: vocab::LineBreak,       // Button.LineBreakMode
    image: text::Text,                  // Button.ImageSource
    is_pressed: bool,                   // Button.IsPressed  (backend writes)
    on_click: fn(*u8, *u8),             // Button.Clicked
    on_click_ctx: *u8,
    on_pressed: fn(*u8, *u8),           // Button.Pressed
    on_pressed_ctx: *u8,
}

const P_LABEL: u64 = 1u64;
const P_TEXT_COLOR: u64 = 2u64;
const P_CORNER_RADIUS: u64 = 4u64;
const P_FONT_SIZE: u64 = 8u64;

import "facet/facet" as facet;
import "facet/button" as button;          // gates the button vocabulary
import "stdlib/option" as option;

struct SaveScreen { saving: bool }

impl SaveScreen: facet::Component {
    fn build(ref this) -> facet::Node {
        var b: facet::Builder = facet::Builder::new();
        b.add(button::Button::new("Save", key: "save")
                 .font_size(13.0f64)
                 .grow(1.0f64)                                  // flex's own modifier
                 .on_click(save_clicked, #addr_of(this) as *u8)
                 .into_node());
        return facet::column(b);
    }
}

fn save_clicked(sender: *u8, cp: *u8) {
    match { (*facet::root_of(cp)).button("save") } {
        option::Option[button::LiveButton]::Some(b) => {
            b.set_label("Saving…");
            b.element().set_is_enabled(false);
        },
        option::Option[button::LiveButton]::None => {},
    };
    return;
}
